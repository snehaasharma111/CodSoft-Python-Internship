print("--- Simple Calculator ---")
a = float(input("1st number: "))
b = float(input("2nd number: "))
op = input("+, -, *, / likho: ")

if op == '+':
    print("Answer =", a+b)
elif op == '-':
    print("Answer =", a-b)
elif op == '*':
    print("Answer =", a*b)
elif op == '/':
    if b == 0:
        print("0 se divide nahi hota!")
    else:
        print("Answer =", a/b)
else:
    print("Galat operator!")
